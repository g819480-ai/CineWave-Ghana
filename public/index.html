// script.js (or catalogue.js) // 1. Fetch the data file you pushed to GitHub async function loadNocturaCatalogue() { try { const response = await fetch('./data/movies.json'); // path to your movies.json if (!response.ok) throw new Error('Failed to load catalogue'); const movies = await response.json(); renderCategories(movies); setupSearch(movies); } catch (error) { console.error('Catalogue error:', error); } } // 2. Group movies by category and render them into your existing containers function renderCategories(movies) { // Map your categories to the container IDs already in your HTML const categories = { 'ghanaian-african': document.getElementById('african-grid'), 'hollywood-action': document.getElementById('hollywood-grid'), 'bollywood': document.getElementById('bollywood-grid'), 'kdrama': document.getElementById('kdrama-grid'), 'international': document.getElementById('international-grid') }; movies.forEach(movie => { const container = categories[movie.category]; if (container) { const card = createMovieCard(movie); container.appendChild(card); } }); } // 3. Create the movie card preserving your existing CSS classes function createMovieCard(movie) { const card = document.createElement('div'); card.className = 'movie-card'; // Use your existing card class card.innerHTML = `
${movie.rightsStatus}

${movie.title}

${movie.year}
`; // Navigate to player or open player modal on click card.addEventListener('click', () => { // Option A: If you use a separate watch page: window.location.href = `watch.html?id=${movie.id}`; // Option B: If you use an inline modal player: // openPlayerModal(movie); }); return card; } document.addEventListener('DOMContentLoaded', loadNocturaCatalogue);
